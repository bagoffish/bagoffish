Two promblems: U have two definitions of one function, so lets just combine them. Second, might as well add a debounce. Something like this:

local debounce = false

function x()

if debounce = false then
debounce = true

print ("Part 1 Functional")
if (script.Parent.BrickColor == BrickColor.new("Bright green"))
then script.Parent.Parent.Brick2.Transparency = 1
script.Parent.Parent.Brick2.CanCollide = false
script.Parent.Parent.Brick5.Transparency = 0
script.Parent.Parent.Brick5.CanCollide = true
script.Parent.BrickColor = BrickColor.new ("Black")
wait (5)
script.Parent.BrickColor = BrickColor.new ("Bright red")

else

if (script.Parent.BrickColor == BrickColor.new ("Bright red"))
print ("Part 2 Functional")

then script.Parent.Parent.Brick2.Transparency = 0
script.Parent.Parent.Brick2.CanCollide = true
script.Parent.Parent.Brick5.Transparency = 1
script.Parent.Parent.Brick5.CanCollide = false
script.Parent.BrickColor = BrickColor.new ("Black")
wait (5)
script.Parent.BrickColor = BrickColor.new ("Bright green")
end -- else

end -- if

debouce = false
end -- function

script.Parent.Touched:connect(x)
